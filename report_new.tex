% --- Thuật ngữ chính ---
\newglossaryentry{fer}{
    name={FER},
    description={\textit{Facial Expression Recognition} -- Nhận dạng biểu cảm khuôn mặt; bài toán tự động phân tích và phân loại biểu cảm khuôn mặt người từ ảnh hoặc video}
}

\newglossaryentry{afea}{
    name={AFEA},
    description={\textit{Automatic Facial Expression Analysis} -- Phân tích biểu cảm khuôn mặt tự động; hệ thống xử lý toàn bộ pipeline từ thu nhận khuôn mặt đến nhận dạng biểu cảm mà không cần can thiệp thủ công}
}

\newglossaryentry{facs}{
    name={FACS},
    description={\textit{Facial Action Coding System} -- Hệ thống mã hóa hành động khuôn mặt; do Ekman và Friesen phát triển năm 1978, mô tả mọi thay đổi có thể quan sát trên khuôn mặt thông qua các đơn vị hành động (AU)}
}

\newglossaryentry{au}{
    name={AU},
    description={\textit{Action Unit} -- Đơn vị hành động; mỗi AU tương ứng với sự co của một cơ mặt hoặc một nhóm cơ xác định. Hệ thống \gls{facs} định nghĩa 44 AU khác nhau}
}

\newglossaryentry{clip}{
    name={CLIP},
    description={\textit{Contrastive Language-Image Pre-training} -- Mô hình nền tảng được Radford et al. (2021) huấn luyện trên hàng trăm triệu cặp ảnh-văn bản từ Internet, có khả năng biểu diễn chung cho cả ngôn ngữ và thị giác}
}

\newglossaryentry{domain_gap}{
    name={domain gap},
    description={(khoảng cách phân phối miền) -- Sự khác biệt về phân phối thống kê giữa tập dữ liệu huấn luyện (source domain) và tập kiểm tra (target domain), dẫn đến sụt giảm hiệu năng khi mô hình gặp dữ liệu từ nguồn mới}
}

\newglossaryentry{zero_shot}{
    name={zero-shot generalization},
    description={(tổng quát hóa không cần mẫu) -- Khả năng của mô hình hoạt động tốt trên domain hoàn toàn chưa thấy trong quá trình huấn luyện, không sử dụng bất kỳ mẫu nào từ target domain}
}

\newglossaryentry{domain_adapt}{
    name={domain adaptation},
    description={(thích nghi miền) -- Kỹ thuật giảm thiểu \gls{domain_gap} bằng cách sử dụng một số mẫu (dù không có nhãn) từ target domain trong quá trình huấn luyện hoặc tinh chỉnh}
}

\newglossaryentry{gabor}{
    name={Gabor wavelet},
    description={(bộ lọc Gabor) -- Bộ lọc kết hợp hàm Gaussian và hàm sinusoidal, nhạy cảm với cả vị trí không gian lẫn tần số không gian. Được dùng phổ biến để trích xuất đặc trưng kết cấu trong ảnh khuôn mặt}
}

\newglossaryentry{geometric_feat}{
    name={geometric feature},
    description={(đặc trưng hình học) -- Nhóm đặc trưng mô tả hình dạng và vị trí của các thành phần khuôn mặt như mắt, lông mày, miệng; thay đổi theo biểu cảm}
}

\newglossaryentry{appearance_feat}{
    name={appearance feature},
    description={(đặc trưng ngoại hình) -- Nhóm đặc trưng mô tả sự thay đổi kết cấu da (texture) khi biểu cảm xuất hiện, ví dụ nếp nhăn và vùng bóng đổ}
}

\newglossaryentry{hmm}{
    name={HMM},
    description={\textit{Hidden Markov Model} -- Mô hình Markov ẩn; mô hình xác suất cho chuỗi thời gian, được dùng để nhận dạng biểu cảm theo chuỗi (sequence-based) nhờ khả năng mô hình hóa dynamics thời gian}
}

\newglossaryentry{svm}{
    name={SVM},
    description={\textit{Support Vector Machine} -- Máy vector hỗ trợ; bộ phân loại tìm siêu phẳng có biên độ lớn nhất để phân tách các lớp, thường dùng kết hợp với đặc trưng Gabor trong \gls{fer}}
}

\newglossaryentry{cnn}{
    name={CNN},
    description={\textit{Convolutional Neural Network} -- Mạng nơ-ron tích chập; kiến trúc học sâu tự động học đặc trưng từ dữ liệu ảnh thông qua các lớp tích chập}
}

\newglossaryentry{vit}{
    name={ViT},
    description={\textit{Vision Transformer} -- Kiến trúc Transformer áp dụng cho thị giác máy tính, chia ảnh thành các patch và xử lý bằng cơ chế self-attention}
}

\newglossaryentry{sigmoid_mask}{
    name={sigmoid mask},
    description={(mặt nạ sigmoid) -- Trong phương pháp CAFE, mặt nạ với giá trị trong $[0,1]$ được học bởi ResNet-18 để chọn lọc các kênh đặc trưng liên quan đến biểu cảm từ CLIP features}
}

\newglossaryentry{hci}{
    name={HCI},
    description={\textit{Human-Computer Interaction} -- Tương tác người-máy; lĩnh vực nghiên cứu và thiết kế giao tiếp giữa con người và hệ thống máy tính}
}

\newglossaryentry{cafe}{
    name={CAFE},
    description={\textit{Cognition of humAn for Facial Expression} -- Phương pháp nhận dạng biểu cảm khuôn mặt có khả năng tổng quát hóa, do Zhang et al. (2024) đề xuất, lấy cảm hứng từ cơ chế nhận thức của con người}
}

\newglossaryentry{kappa}{
    name={coefficient kappa},
    description={(hệ số kappa) -- Thước đo mức độ nhất quán giữa hai người quan sát (inter-observer agreement), hiệu chỉnh phần trùng khớp ngẫu nhiên; khuyến nghị dùng thay cho tỉ lệ chính xác thô khi đánh giá độ tin cậy ground truth}
}

% ============= ABSTRACT =============
\begin{abstract}
Báo cáo này trình bày nghiên cứu về bài toán \gls{fer}, một lĩnh vực quan trọng trong thị giác máy tính và \gls{hci}. Phần lý thuyết tập trung phân tích Chapter~19 ``Facial Expression Recognition'' từ \textit{Handbook of Face Recognition} (2nd Edition) của Tian, Kanade và Cohn (2011), bao gồm kiến trúc tổng quát của một hệ thống \gls{afea}, từ thu nhận khuôn mặt đến trích xuất đặc trưng và nhận dạng biểu cảm. Bài báo SOTA được chọn là \textit{``Generalizable Facial Expression Recognition''} (\gls{cafe}, Zhang et al., 2024), đề xuất cơ chế học \gls{sigmoid_mask} trên đặc trưng khuôn mặt tổng quát từ \gls{clip}, nhằm giải quyết \gls{zero_shot} -- hạn chế căn bản mà các phương pháp truyền thống chưa đề cập đến. Thực nghiệm được thực hiện trên năm bộ dữ liệu, cho thấy \gls{cafe} vượt trội các phương pháp SOTA hiện tại với biên độ đáng kể trên các tập kiểm tra có \gls{domain_gap}.

\textbf{Từ khóa:} Facial Expression Recognition, Action Units, FACS, CLIP, Sigmoid Mask, Domain Generalization, Zero-shot Generalization, CAFE.
\end{abstract}

\section{PHẦN LÝ THUYẾT}

\subsection{Nghiên cứu chương sách: Chapter 19 -- Facial Expression Recognition}

\subsubsection{Giới thiệu chương}

Chapter~19 trong cuốn \textit{Handbook of Face Recognition} (2nd Edition, Springer, 2011), được biên soạn bởi Yingli Tian (The City College of New York), Takeo Kanade (Carnegie Mellon University) và Jeffrey F. Cohn (University of Pittsburgh), là một trong những tổng quan toàn diện và có ảnh hưởng nhất về \gls{fer} tính đến thời điểm đó.

Theo chương sách, \textit{facial expression analysis} được định nghĩa là các hệ thống máy tính tự động phân tích và nhận dạng chuyển động khuôn mặt từ thông tin hình ảnh. Cần phân biệt rõ khái niệm này với \textit{emotion analysis}: biểu cảm khuôn mặt là tín hiệu mang tính trung lập về ngữ nghĩa hơn, phản ánh không chỉ cảm xúc mà còn ý định, quá trình nhận thức và ý nghĩa giao tiếp khác. Một nụ cười (\gls{au} 12) có thể là niềm vui, nhưng cũng có thể là phép lịch sự xã giao hay che giấu cảm xúc thật. Phân tích cảm xúc đòi hỏi tri thức cao hơn nhiều, bao gồm ngữ cảnh, giọng nói và cử chỉ cơ thể -- những yếu tố nằm ngoài phạm vi của một hệ thống \gls{afea} thuần túy.

Chương tổ chức nội dung xung quanh bốn vấn đề kỹ thuật chính: thu nhận khuôn mặt (face acquisition), trích xuất và biểu diễn đặc trưng (\gls{geometric_feat} và \gls{appearance_feat}), nhận dạng biểu cảm, và phân tích đa phương thức. Ngoài ra còn có phân tích không gian bài toán với 10 chiều thách thức và 7 câu hỏi mở định hướng cho nghiên cứu tương lai.

\subsubsection{Kiến trúc tổng quát của hệ thống AFEA}

Một hệ thống \gls{afea} hoàn chỉnh được cấu thành từ ba giai đoạn tuần tự.

\begin{figure}[H]
\begin{minipage}[t]{0.62\textwidth}

    \includegraphics[width=\linewidth]{fig_afea_pipeline.png}
\end{minipage}%

\textbf{Giai đoạn 1 -- Face Acquisition.}
Giai đoạn này tự động tìm và định vị vùng khuôn mặt trong ảnh hoặc chuỗi video. Có hai chiến lược phổ biến. Chiến lược thứ nhất là phát hiện từng frame một (per-frame detection), phù hợp khi không giả định gì về chuyển động đầu; Viola-Jones detector \cite{viola2001} là lựa chọn tiêu biểu cho trường hợp này, vì có thể chạy real-time. Chiến lược thứ hai chỉ phát hiện khuôn mặt ở frame đầu tiên rồi dùng tracking cho các frame tiếp theo, tiết kiệm chi phí tính toán nhưng dễ trượt khi có chuyển động đầu lớn. Với chuyển động ngoài mặt phẳng (out-of-plane rotation), cần thêm bước ước lượng tư thế đầu (head pose estimation) trước khi trích xuất đặc trưng.

\textbf{Giai đoạn 2 -- Feature Extraction.}
Sau khi khuôn mặt được định vị, hệ thống phân tích những thay đổi do biểu cảm gây ra theo hai hướng song song: \gls{geometric_feat} (hình dạng và vị trí các thành phần khuôn mặt) và \gls{appearance_feat} (thay đổi kết cấu da). Hai nhóm này sẽ được phân tích chi tiết trong Mục~\ref{subsec:feature_extraction}.

\textbf{Giai đoạn 3 -- Expression Recognition.}
Tùy mục tiêu, hệ thống nhận dạng ở một trong hai mức: mức \gls{au} (nhận dạng từng đơn vị hành động và tổ hợp của chúng -- chi tiết nhất, phù hợp ứng dụng y tế và tâm lý), hoặc mức nhãn cảm xúc (phân loại thành 6 biểu cảm cảm xúc cơ bản -- trực quan hơn, phù hợp \gls{hci}). Theo cách xử lý thời gian, có hai nhóm phương pháp: frame-based (phân tích từng ảnh độc lập) và sequence-based (khai thác dynamics của chuỗi frame).

\subsubsection{Hệ thống FACS và Action Units}

\textbf{Nền tảng tâm lý học.}
Nghiên cứu biểu cảm khuôn mặt có lịch sử từ năm 1872 khi Darwin xuất bản \textit{The Expression of the Emotions in Man and Animals}. Đến thập niên 1970, Paul Ekman và Wallace Friesen đề xuất rằng có 6 cảm xúc cơ bản -- disgust (ghê tởm), fear (sợ hãi), joy (vui mừng), surprise (ngạc nhiên), sadness (buồn bã), anger (tức giận) -- được biểu hiện qua cùng một tập biểu cảm khuôn mặt trên toàn thế giới, bất kể văn hóa.

Trong thực tế đời thường, các biểu cảm nguyên mẫu này xuất hiện tương đối hiếm. Cảm xúc thường được truyền đạt qua những thay đổi tinh tế ở một vài đặc trưng riêng lẻ, ví dụ môi khép chặt khi không thoải mái hay nhíu lông mày nhẹ khi tập trung. Để mã hóa sự tinh tế đó, Ekman và Friesen (1978) phát triển \gls{facs}.

\textbf{Định nghĩa và cấu trúc FACS.}
\gls{facs} là hệ thống mô tả tất cả các thay đổi có thể quan sát được trên khuôn mặt thông qua các \gls{au}, mỗi \gls{au} tương ứng với sự co của một cơ mặt cụ thể. Đây là hệ thống thuần túy mô tả, không gán nhãn cảm xúc. \gls{facs} bao gồm 44 \gls{au}, chia thành 30 \gls{au} giải phẫu học và 14 \gls{au} miscellaneous. Một số \gls{au} quan trọng:

    \item \textbf{AU1} -- Inner Brow Raise: nâng phần trong lông mày (cơ frontalis, pars medialis).
    \item \textbf{AU4} -- Brow Lowerer: cau lông mày (cơ corrugator supercilii).
    \item \textbf{AU6} -- Cheek Raiser: nâng gò má (cơ orbicularis oculi, pars orbitalis).
    \item \textbf{AU12} -- Lip Corner Puller: kéo góc môi lên, tạo nụ cười (cơ zygomaticus major).
    \item \textbf{AU25} -- Lips Part: mở môi (cơ depressor labii).
\end{itemize}

\textbf{Cường độ và tổ hợp AU.}
Các \gls{au} biến đổi về cường độ theo thang 5 điểm từ A (trace -- vết mờ) đến E (extreme -- cực mạnh). Trong thực tế, các \gls{au} thường xuất hiện kết hợp theo hai dạng. Tổ hợp cộng tính (additive combinations): các \gls{au} kết hợp mà không ảnh hưởng lẫn nhau, ví dụ nụ cười mở miệng là AU12 kết hợp với AU25/AU26/AU27. Tổ hợp phi cộng tính (non-additive combinations hay co-articulation effects): các \gls{au} tương tác và ảnh hưởng đến ngoại hình của nhau, ví dụ AU12 kết hợp AU15 thường xuất hiện khi ngượng ngùng, kết quả không thể dự đoán đơn giản bằng cách cộng từng \gls{au}. Bảng~\ref{tab:au_combinations} minh họa một số tổ hợp \gls{au} điển hình theo EMFACS.

\subsubsection{Không gian vấn đề}

Chương sách xác định bài toán \gls{fer} không phải là một chiều duy nhất mà là không gian đa chiều với 10 nhân tố phức tạp. Mỗi nhân tố đặt ra thách thức riêng và một hệ thống \gls{afea} lý tưởng phải đối phó được với tất cả.

\textbf{(1) Mức độ mô tả.} Có hai cực: nhận dạng 6 cảm xúc nguyên mẫu của Ekman (coarse-grained) và nhận dạng từng \gls{au} cùng tổ hợp (fine-grained). Nhận dạng ở mức \gls{au} chi tiết hơn nhưng đối mặt với không gian tổ hợp cực lớn (7000+ tổ hợp).

\textbf{(2) Khác biệt cá nhân.} Hình dạng, kết cấu da, màu sắc và lông mặt biến đổi theo giới tính, chủng tộc và độ tuổi. Ngoài khác biệt ngoại hình, còn có khác biệt về expressiveness -- mức độ linh hoạt cơ mặt và tần suất biểu cảm mạnh.

\textbf{(3) Chuyển tiếp giữa các biểu cảm.} Trong thực tế, các \gls{au} có thể chuyển thẳng từ tổ hợp này sang tổ hợp khác mà không qua trạng thái trung lập, đòi hỏi dữ liệu huấn luyện phải bao gồm các chuyển tiếp động học.

\textbf{(4) Cường độ biểu cảm.} Một phương pháp hoạt động tốt với biểu cảm cường độ cao có thể tổng quát kém sang biểu cảm cường độ thấp, trong khi phần lớn biểu cảm tự nhiên lại ở mức thấp đến trung bình.

\textbf{(5) Biểu cảm dàn dựng vs. tự nhiên.} Biểu cảm dàn dựng và tự nhiên được điều khiển bởi hai con đường thần kinh khác nhau: pyramidal (có ý thức) và extrapyramidal (không ý thức). Kiểm soát cơ mặt của biểu cảm dàn dựng thường kém chính xác và kém đối xứng hơn.

\textbf{(6) Tư thế đầu và độ phức tạp cảnh nền.} Phần lớn hệ thống \gls{fer} giả định khuôn mặt nhìn thẳng (frontal), trong khi thực tế chuyển động đầu lớn rất phổ biến và thường xảy ra đồng thời với biểu cảm.

\textbf{(7) Thu nhận ảnh và độ phân giải.} Theo thực nghiệm trong chương sách, tại $48 \times 64$ pixel, các đặc trưng như góc mắt và miệng bắt đầu khó phát hiện; dưới $24 \times 32$ pixel, biểu cảm gần như không nhận dạng được.

\subsubsection{Thu nhận khuôn mặt}

Với khuôn mặt nhìn gần thẳng, Viola và Jones (2001) phát triển bộ phát hiện real-time dựa trên tập đặc trưng hình chữ nhật (rectangle features) và AdaBoost, là phương pháp được dùng rộng rãi nhất trong các hệ thống \gls{afea} thời kỳ này. Để xử lý chuyển động đầu lớn ngoài mặt phẳng, có hai hướng tiếp cận chính.

\textit{Phương pháp dựa trên mô hình 3D.} Xiao et al. (2003) dùng cylindrical head model kết hợp với Active Appearance Model (AAM), cho phép ước lượng chuyển động 6 bậc tự do (DOF). Hệ thống tự động chọn các frame tham chiếu để hiệu chỉnh sai số tích lũy, đạt sai số tuyệt đối trung bình khoảng 3.86 độ với yaw tối đa 75 độ và pitch 40 độ.

\subsubsection{Trích xuất và Biểu diễn Đặc trưng}
\label{subsec:feature_extraction}

\textbf{(1) Geometric Features.}

\gls{geometric_feat} mô tả hình dạng và vị trí của các thành phần khuôn mặt. Tian et al. (2001) phát triển multi-state models để theo dõi và mã hóa trạng thái từng thành phần: môi (3 trạng thái: mở, đóng, khép chặt), mỗi mắt (2 trạng thái: mở, nhắm), lông mày và má (1 tham số liên tục). Hệ thống tổng hợp 15 tham số cho vùng mặt trên và 9 tham số cho vùng mặt dưới, tất cả chuẩn hóa theo ảnh trung lập để loại bỏ sự khác biệt về kích thước khuôn mặt.

Ngoài permanent features, còn có transient features -- các đặc trưng thoáng hiện chỉ xuất hiện khi có biểu cảm cụ thể, ví dụ nếp nhăn nasolabial (rãnh mũi-miệng) và nếp nhăn đuôi mắt, được mã hóa theo trạng thái nhị phân.

\textbf{(2) Appearance Features.}

\gls{appearance_feat} mô tả sự thay đổi kết cấu da mà đặc trưng hình học không nắm bắt được. \gls{gabor} là phương pháp trích xuất phổ biến nhất, được định nghĩa bởi:

\begin{equation}
    G(x, y; f, \theta) = \frac{f^2}{\pi \gamma \eta} \exp\!\left(-\frac{f^2}{\gamma^2}x'^2 - \frac{f^2}{\eta^2}y'^2\right) \exp(2\pi i f x')
    \label{eq:gabor}
\end{equation}

trong đó $x' = x\cos\theta + y\sin\theta$, $y' = -x\sin\theta + y\cos\theta$, $f$ là tần số không gian và $\theta$ là hướng. Bộ lọc Gabor phát hiện các pattern texture ở nhiều tần số và hướng, tương tự cơ chế xử lý của visual cortex sơ cấp.

Tian et al. (2002) dùng bộ lọc Gabor với 5 tần số không gian $\times$ 8 hướng $= 40$ bộ lọc, áp dụng tại các vị trí cố định trên khuôn mặt: 480 hệ số cho vùng mặt trên và 432 hệ số cho vùng mặt dưới.

Tian et al. (2002) thực hiện so sánh toàn diện trên Cohn-Kanade database: \gls{gabor} đơn lẻ hoạt động tốt cho \gls{au} đơn giản trên tập dữ liệu đồng nhất, nhưng kém khi gặp tổ hợp \gls{au} và đối tượng đa dạng; \gls{geometric_feat} đơn lẻ nhất quán hơn; kết hợp cả hai cải thiện nhận dạng cho tất cả \gls{au}, đặc biệt là tổ hợp phức tạp.

\subsubsection{Nhận dạng Biểu cảm}

\textbf{Phương pháp frame-based.}
Tian et al. (2001) dùng mạng nơ-ron 3 lớp với back-propagation, đây là hệ thống đầu tiên xử lý tổ hợp \gls{au} bằng cách cho phép nhiều output node kích hoạt đồng thời, đạt 95.5\% trên neutral + 16 \gls{au}. Bartlett et al. (2002) dùng \gls{svm} với \gls{gabor} features kết hợp Multinomial Logistic Regression, đạt 91.5\%. Yang et al. (2009) chuyển bài toán nhận dạng và ước lượng cường độ thành ranking problem với RankBoost, đạt 88\% trên 6 cảm xúc.

\textbf{Phương pháp sequence-based.}
Bartlett et al. (2006) dùng \gls{svm} để trích xuất đặc trưng Gabor rồi dùng \gls{hmm} mô hình hóa dynamics thời gian, đạt 98.1\% cho blink/non-blink. Cohn et al. (2002) dùng bộ phân loại dựa quy tắc cho eye và brow \gls{au} trong hành vi spontaneous, đạt 98\% tổng thể cho 3 hành vi mắt.

\subsubsection{Bảy Câu hỏi Mở}

\textbf{Q1.} Con người nhận dạng biểu cảm theo cơ chế nào? Bằng cách so sánh nhận dạng của con người và máy, ta có thể cải thiện cả hai.

\textbf{Q2.} Phân tích ở mức độ chi tiết hơn có luôn tốt hơn không? Câu trả lời phụ thuộc vào chất lượng ảnh và loại ứng dụng; cần hệ thống ``self-aware'' biết mức độ nhận dạng khả thi dựa trên chất lượng đầu vào.

\textbf{Q3.} Có cách mã hóa biểu cảm tốt hơn \gls{facs} không? \gls{facs} là hệ thống human-observer-based với khả năng phân biệt cường độ hạn chế, cần hệ thống mã hóa định lượng hơn.

\textbf{Q4.} Làm sao thu được ground truth đáng tin cậy? Cần công bố tiêu chí gán nhãn và inter-observer agreement, đo bằng \gls{kappa}.

\textbf{Q5.} Làm sao nhận dạng biểu cảm trong điều kiện thực tế? Đây chính xác là vấn đề mà bài báo \gls{cafe} (2024) hướng đến giải quyết sau 13 năm.

\textbf{Q6.} Làm sao khai thác tốt thông tin thời gian? Biểu cảm tự nhiên và dàn dựng có thể phân biệt qua timing parameters.

\textbf{Q7.} Làm sao tích hợp \gls{fer} với các phương thức khác để giảm nhập nhằng?

\subsection{Từ tiếp cận thủ công đến học sâu}

Bài toán \gls{fer} trải qua hơn 40 năm phát triển, chia thành ba giai đoạn chính.

\textbf{Giai đoạn 1 -- Handcrafted Features và Rule-based Systems (trước 2012).}
Giai đoạn của chương sách nhóm nghiên cứu. Đặc trưng được thiết kế thủ công bởi con người (\gls{gabor}, LBP, HOG, optical flow) và bộ phân loại truyền thống (\gls{svm}, mạng nơ-ron nông, \gls{hmm}). Dữ liệu nhỏ, thu thập trong phòng lab, biểu cảm dàn dựng. Tỉ lệ nhận dạng đạt 80--95\% trên tập kiểm tra cùng điều kiện với tập huấn luyện, nhưng thất bại nặng khi điều kiện thay đổi. Tian et al. (2001) với hệ thống \gls{geometric_feat} + \gls{gabor} và mạng nơ-ron đạt 95.5\% trên 16 \gls{au} là milestone quan trọng nhất giai đoạn này.

\textbf{Giai đoạn 2 -- Deep Learning và In-the-Wild Datasets (2013--2020).}
AlexNet (2012) đánh dấu bước ngoặt. \gls{cnn} tự động học đặc trưng từ dữ liệu, không cần thiết kế thủ công. Các dataset in-the-wild bùng nổ: FER2013 (35K ảnh từ Google Images), AffectNet (450K ảnh), RAF-DB (15K ảnh). Kỹ thuật xử lý overfitting và label noise trở thành vấn đề trung tâm.

\textbf{Giai đoạn 3 -- Foundation Models và Generalization (2021 đến nay).}
\gls{vit}, \gls{clip} và các large pre-trained models mở ra hướng đi mới: thay vì train từ đầu, tận dụng tri thức từ mô hình được tiền huấn luyện trên dữ liệu khổng lồ. Vấn đề \gls{domain_gap} trở thành trọng tâm. \gls{cafe} (2024) là đại diện tiêu biểu của giai đoạn này.

\textbf{Bài toán mới: Zero-shot Generalization (GFER).} Zhang et al. đặt ra bài toán Generalizable \gls{fer}: train chỉ trên một dataset duy nhất, không sử dụng bất kỳ mẫu nào từ target domain, nhưng phải nhận dạng tốt trên nhiều test set có \gls{domain_gap} khác nhau.

\textbf{Thành phần 1 -- Fixed CLIP Face Features.}
\gls{clip} ViT-B/32 visual encoder trích xuất $\mathbf{F} \in \mathbb{R}^{N \times C}$ ($C = 512$), được đóng băng hoàn toàn trong suốt training để bảo toàn khả năng generalization. Gradient không được phép cập nhật \gls{clip} weights.

\textbf{Thành phần 2 -- Sigmoid Mask Learning.}
ResNet-18 học mask $\mathbf{M}$ để chọn lọc các kênh đặc trưng liên quan đến biểu cảm từ \gls{clip} features:

\begin{equation}
    \mathbf{M}_s = \text{Sigmoid}(\mathbf{M}), \quad \mathbf{M}_s \in [0,1]^{N \times C}
    \label{eq:sigmoid_mask}
\end{equation}

Hàm sigmoid ép giá trị mask về $[0, 1]$, mang ý nghĩa xác suất chọn từng kênh đặc trưng -- tạo ra dạng soft channel attention. Đặc trưng sau chọn lọc:

\begin{equation}
    \tilde{\mathbf{F}} = \mathbf{M}_s \odot \mathbf{F}
    \label{eq:masked_feature}
\end{equation}

\textbf{Thành phần 3 -- Channel Separation và Channel-Diverse Loss.}
$\tilde{\mathbf{F}}$ được chia thành $L=7$ phần theo chiều kênh, mỗi phần tương ứng một biểu cảm cơ bản. Max Pooling thay cho FC layer để tạo logits:

\begin{equation}
    \bar{\mathbf{F}}^d = \left\{\max(\tilde{\mathbf{F}}_1 \odot \mathbf{M}^{drop}_1),\ \ldots,\ \max(\tilde{\mathbf{F}}_L \odot \mathbf{M}^{drop}_L)\right\}
    \label{eq:channel_sep}
\end{equation}

Channel-diverse loss buộc mask của các biểu cảm khác nhau phải thực sự khác nhau:

\begin{equation}
    l_{div} = 1 - \frac{1}{Nc} \sum_{i=1}^{N} \sum_{j=1}^{L} \tilde{\mathbf{F}}_{max_{ij}}
    \label{eq:div_loss}
\end{equation}

Tổng loss training kết hợp ba thành phần:

\begin{equation}
    l_{train} = l_{cls} + \lambda\, l_{sep} + \beta\, l_{div}, \quad \lambda=1.5,\ \beta=5
    \label{eq:total_loss}
\end{equation}

Khi inference, chỉ giữ module tính $l_{cls}$; hai loss còn lại chỉ hoạt động như regularizer trong training.

\subsection{Phân tích và Đánh giá}

\textbf{Mối liên hệ kỹ thuật với chương sách.}
Chương sách dùng \gls{gabor} -- handcrafted features dựa trên tri thức sinh học thị giác. \gls{cafe} dùng \gls{clip} -- learned features từ dữ liệu khổng lồ. Cả hai đều hướng đến đặc trưng có khả năng generalize, nhưng theo hai triết lý trái ngược: thiết kế để tổng quát vs. học để tổng quát.

\centering

\begin{tabular}{|l|p{5cm}|p{5cm}|}
\hline
\textbf{Khía cạnh} & \textbf{Chương sách (2011)} & \textbf{\gls{cafe} (2024)} \\ \hline
Feature extraction & Handcrafted: \gls{gabor} + \gls{geometric_feat} & Learned: \gls{clip} ViT-B/32 \\ \hline

\end{tabular}

\label{tab:comparison}
\end{table}

\textbf{Điểm mạnh kỹ thuật.} \gls{cafe} giải quyết \gls{zero_shot} một cách có hệ thống mà không cần dữ liệu target domain. Thiết kế đơn giản và interpretable: \gls{sigmoid_mask} có thể visualize và giải thích trực quan. Flexible với nhiều backbone (MobileNet, ResNet-18, ResNet-50). Kết quả so sánh với CLIP+Finetune (baseline dùng \gls{clip} fixed + FC layer, cùng số trainable params) cho thấy \gls{cafe} đạt 63.48\% vs 55.73\% mean accuracy, chứng minh hiệu năng đến từ thiết kế phương pháp, không đơn thuần từ việc dùng \gls{clip}.

\textbf{Hạn chế còn lại.} \gls{cafe} chỉ xử lý ảnh tĩnh, không khai thác temporal dynamics mà chương sách đã nhấn mạnh từ 2011 qua \gls{hmm}. Accuracy trên AffectNet vẫn thấp (45.86\%) do label noise cao. Phương pháp giới hạn ở 7 biểu cảm cơ bản, chưa xử lý mức \gls{au} hay compound expressions. \gls{clip} có bias về chủng tộc và văn hóa, và \gls{cafe} thừa hưởng bias này.

\textbf{Đề xuất hướng cải tiến.} Kết hợp \gls{cafe} với LSTM hoặc Temporal Transformer để xử lý video sequence; áp dụng channel separation cho 44 \gls{au} thay vì 7 lớp; thay \gls{clip} bằng face-specific foundation model (ArcFace, FaRL); kết hợp với noise-robust label learning cho AffectNet.

\section{PHẦN THỰC HÀNH VÀ DEMO MINH HỌA}

\subsection{Phân tích kỹ thuật từ mã nguồn}

Nhóm phân tích mã nguồn gốc tại \url{https://github.com/zyh-uaiaaaa/Generalizable-FER} và thực hiện reproduce độc lập. Bốn thành phần kỹ thuật cốt lõi được xác định.

\textbf{(1) CLIP đóng băng hoàn toàn.}
\gls{clip} ViT-B/32 được load và tất cả gradient bị chặn qua \texttt{p.requires\_grad = False}. Trong mỗi forward pass, ảnh đầu vào được encode thành $\mathbf{F} \in \mathbb{R}^{N \times 512}$ với \texttt{torch.no\_grad()}. Nếu cho phép fine-tune \gls{clip}, mô hình sẽ overfit vào domain training và mất khả năng \gls{zero_shot}.

\textbf{(2) Sigmoid Mask Learning (ResNet-18).}
ResNet-18 (pre-train trên MS-Celeb) học \gls{sigmoid_mask} $\mathbf{M}$ từ ảnh đầu vào. Mask được áp dụng vào \gls{clip} features qua: $\tilde{\mathbf{F}} = \sigma(\mathbf{M}) \odot \mathbf{F}$. Hàm sigmoid ép giá trị về $[0,1]$, mang ý nghĩa xác suất chọn từng kênh. Thay bằng ReLU hoặc Linear thì mask có thể nhận giá trị tùy ý và overfit mạnh hơn. Đây là thành phần có đóng góp lớn nhất trong ablation study (+8.82\%).

\textbf{(3) Channel Separation và Channel-Diverse Loss.}
Hàm \texttt{supervisor()} thực hiện hai thao tác song song. $L_{div}$: $\tilde{\mathbf{F}}$ được reshape và MaxPool chia thành 7 nhóm, loss buộc các nhóm phải khác nhau tối đa để tránh học cùng pattern. $L_{sep}$: áp dụng channel-drop mask ngẫu nhiên (63/73 kênh mỗi lần), buộc mỗi nhóm kênh phải tự đủ để nhận dạng biểu cảm tương ứng, tăng redundancy và robustness.

\textbf{(4) Hàm Loss tổng hợp.}
\begin{equation}
    \mathcal{L}_{train} = \mathcal{L}_{cls} + 5.0 \cdot \mathcal{L}_{div} + 1.5 \cdot \mathcal{L}_{sep}
\end{equation}
Trọng số $\beta = 5.0$ cho $\mathcal{L}_{div}$ lớn hơn $\lambda = 1.5$ cho $\mathcal{L}_{sep}$, thể hiện ưu tiên diversity giữa các nhóm hơn tính tự-đủ của từng nhóm. Khi inference, chỉ dùng $\mathcal{L}_{cls}$.

\subsubsection{EXP-IMPROVE03: Cải tiến CAFE + FreqAug + L\_sem}

Nhóm đề xuất hai cải tiến bổ trợ nhau để giải quyết domain shift ở input level, bổ sung cho feature-level generalization của \gls{cafe} gốc.

Cải tiến đầu tiên là Frequency Augmentation (FreqAug), gồm ba kỹ thuật áp dụng trên miền tần số trước khi đưa ảnh vào mạng: PASTA (thêm nhiễu ngẫu nhiên vào biên độ FFT, giữ nguyên phase), AAD (mask 15\% hệ số tần số ngẫu nhiên), và AAUA (thêm nhiễu Gaussian nhỏ vào biên độ FFT). Augmentation trên frequency domain hiệu quả hơn pixel domain vì thay đổi amplitude ảnh hưởng texture/style (domain-specific) trong khi phase bảo toàn nội dung biểu cảm (domain-invariant).

Cải tiến thứ hai là Semantic Alignment Loss ($L_{sem}$), kéo masked features $\tilde{\mathbf{F}}$ về phía không gian ngữ nghĩa của \gls{clip} thông qua 5 prompt templates per emotion:
\begin{equation}
    \mathcal{L}_{sem} = -\frac{1}{N} \sum_{i=1}^{N} \log \frac{\exp(\tilde{\mathbf{F}}_i \cdot \mathbf{T}_{y_i} / \tau)}{\sum_{j=1}^{7} \exp(\tilde{\mathbf{F}}_i \cdot \mathbf{T}_j / \tau)}
\end{equation}

Loss mới: $\mathcal{L} = \mathcal{L}_{cls} + 5.0 \cdot \mathcal{L}_{div} + 1.5 \cdot \mathcal{L}_{sep} + 2.0 \cdot \mathcal{L}_{sem}$.

% ============= SECTION 6: KẾT LUẬN =============
\section{KẾT LUẬN}

Chương sách cung cấp nền tảng lý thuyết vững chắc cho bài toán \gls{fer}: từ hệ thống \gls{facs} để mô tả biểu cảm ở mức \gls{au}, đến pipeline \gls{afea} ba bước, đến các kỹ thuật cụ thể như \gls{gabor} và \gls{hmm}. Dù các kỹ thuật này đã được học sâu thay thế, triết lý thiết kế và không gian vấn đề mà chương sách xác định vẫn còn nguyên giá trị.

Bài báo \gls{cafe} giải quyết câu hỏi mở quan trọng nhất mà chương sách đặt ra từ năm 2011 -- nhận dạng biểu cảm trong điều kiện thực tế -- bằng cách xây dựng đặc trưng đủ tổng quát ngay từ đầu thay vì thích nghi sang domain mới. Kết quả vượt trội của \gls{cafe} trên tất cả cặp train/test cross-domain cho thấy hướng đi này có tiềm năng lớn.

Nhiều câu hỏi từ 2011 vẫn chưa có lời giải hoàn toàn: nhận dạng \gls{au} mịn hơn, xử lý biểu cảm tự nhiên trong video, và ground truth đáng tin cậy theo \gls{kappa} vẫn là những thách thức mở cho tương lai.
